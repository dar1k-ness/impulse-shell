<?php
declare(strict_types=1);
namespace App\Models;

use InvalidArgumentException as Bad;

/** A user. */
final class User extends Model implements \JsonSerializable
{
    public const LIMIT = 0x1F;
    private static int $count = 0;

    public function __construct(private string $name, protected ?int $age = null) {}

    public function greet(string ...$to): string
    {
        if ($this->age === null || $this->age < 18) { throw new Bad('too young'); }
        $list = array_map(fn($x) => strtoupper($x), $to);
        return "Hello, {$this->name} and $list[0]!\n" . 'raw $x' . 1.5e3;
    }
}
