cmake_minimum_required(VERSION 3.20)
project(demo VERSION 1.2.3 LANGUAGES C CXX)
# options
option(BUILD_TESTS "Build tests" ON)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Quick)
add_executable(${PROJECT_NAME} src/main.cpp "src/a b.cpp")
target_link_libraries(${PROJECT_NAME} PRIVATE Qt6::Core Qt6::Quick)
if(BUILD_TESTS AND NOT WIN32)
  enable_testing()
  add_subdirectory(tests)
elseif(DEFINED ENV{CI})
  message(STATUS "CI build: $ENV{CI} ${CMAKE_BUILD_TYPE}")
endif()
foreach(f IN LISTS SOURCES)
  message("${f}")
endforeach()
install(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION bin)
