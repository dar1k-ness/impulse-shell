/** @param {string} id */
export async function getUser(id, opts = {}) {
  const res = await fetch(`https://api.example.com/users/${id}?v=${1 + 2}`);
  if (!res.ok) throw new Error('HTTP ' + res.status);
  let n = 0x1f + 1.5e3 + 10n;
  return { ...(await res.json()), n, ok: true, x: null, y: undefined };
}
