-- users and orders
CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(64) NOT NULL DEFAULT 'anon',
    balance NUMERIC(10, 2) CHECK (balance >= 0.00),
    created_at TIMESTAMP DEFAULT now()
);
/* report */
SELECT u.name, COUNT(o.id) AS orders, SUM(o.total) * 1.2 AS "gross"
FROM users u
LEFT JOIN orders o ON o.user_id = u.id AND o.status IN ('paid', 'sent')
WHERE u.created_at BETWEEN '2026-01-01' AND CURRENT_DATE
  AND u.name LIKE 'A%' OR u.id <> 42
GROUP BY u.name HAVING COUNT(o.id) > 3
ORDER BY orders DESC LIMIT 10;
UPDATE users SET balance = balance - 5 WHERE id = 1;
