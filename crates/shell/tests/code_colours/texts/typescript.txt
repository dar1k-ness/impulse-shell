export abstract class Cache<K extends Key, V> {
  private readonly items = new Map<K, Entry<V>>();
  constructor(protected ttl: number = 60_000) {}
  get(key: K): V | undefined {
    const e = this.items.get(key);
    return e && (e.expires ?? Infinity) > Date.now() ? e.value : undefined;
  }
  abstract load(key: K): Promise<V>;
}
