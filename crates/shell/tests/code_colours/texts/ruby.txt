# frozen_string_literal: true
require 'json'

module Demo
  class User < Base
    attr_reader :name, :age
    LIMIT = 0x1F

    def initialize(name, age = 18)
      @name, @age = name, age
      @@count ||= 0
    end

    def adult? = age >= 18

    def to_s
      "#{name} (#{age + 1}) #{'x' * 3}\n"
    end
  end
end

users = [Demo::User.new("Ann", 30), Demo::User.new(:bob.to_s)]
users.select(&:adult?).each_with_index do |u, i|
  puts "#{i}: #{u}" unless u.name =~ /^b\w+$/i
end
h = { key: 1, "str" => 2.5e3, sym: :val, nothing: nil }
puts h.to_json if $stdout.tty? && !h.empty?
